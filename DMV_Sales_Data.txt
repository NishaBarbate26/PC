import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns

# Step 1: Create sample sales datasets in CSV, Excel and JSON formats

csv_data = pd.DataFrame([
    [101, "2026-01-05", "Amit", "Laptop", "Electronics", 2, 50000, "Pune"],
    [102, "2026-01-10", "Neha", "Chair", "Furniture", 4, 2500, "Mumbai"],
    [103, "2026-02-12", "Raj", "Phone", "electronics", 1, 20000, "Pune"],
    [104, "2026-02-15", "Sara", "Table", "Furniture", np.nan, 6000, "Delhi"],
    [105, "2026-03-02", "Ravi", "Watch", "Accessories", 3, 1500, "Mumbai"]
], columns=["OrderID", "Date", "Customer", "Product", "Category",
            "Quantity", "UnitPrice", "Region"])

excel_data = pd.DataFrame([
    [103, "2026-02-12", "Raj", "Phone", "Electronics", 1, 20000, "Pune"],
    [106, "2026-03-08", "Priya", "Laptop", "Electronics", 1, 55000, "Delhi"],
    [107, "2026-03-12", "Aman", "Sofa", "Furniture", 2, 15000, "Pune"],
    [108, "2026-04-01", "Isha", "Headphones", "Accessories", 5, 1000, "Mumbai"],
    [109, "2026-04-05", "Kiran", "Phone", "Electronics", 2, np.nan, "Delhi"]
], columns=csv_data.columns)

json_data = pd.DataFrame([
    [110, "2026-04-10", "Meera", "Desk", "Furniture", 2, 8000, "Pune"],
    [111, "2026-04-15", "Arjun", "Watch", "Accessories", 4, 2000, "Mumbai"],
    [112, "2026-05-02", "Riya", "Laptop", "Electronics", 1, 60000, "Delhi"],
    [113, "2026-05-10", "Dev", "Chair", "Furniture", 3, 3000, "Pune"],
    [114, "2026-05-15", "Anaya", "Phone", "Electronics", 2, 18000, "Mumbai"]
], columns=csv_data.columns)

# Save the datasets in their respective file formats
csv_data.to_csv("sales.csv", index=False)
excel_data.to_excel("sales.xlsx", index=False)
json_data.to_json("sales.json", orient="records", indent=4)

# Step 2: Load data from all three file formats
df_csv = pd.read_csv("sales.csv")
df_excel = pd.read_excel("sales.xlsx")
df_json = pd.read_json("sales.json")

print("\n--- CSV DATA ---")
print(df_csv.head())
print("\n--- EXCEL DATA ---")
print(df_excel.head())
print("\n--- JSON DATA ---")
print(df_json.head())

# Step 3: Explore data and identify missing values and duplicates
print("\nCSV information:")
df_csv.info()
print("\nMissing values in each file:")
print("CSV:\n", df_csv.isnull().sum())
print("Excel:\n", df_excel.isnull().sum())
print("JSON:\n", df_json.isnull().sum())

# Step 4: Combine all datasets into one DataFrame
df = pd.concat([df_csv, df_excel, df_json], ignore_index=True)

print("\nCombined dataset shape:", df.shape)
print("\nDuplicate rows:", df.duplicated().sum())
print("Duplicate Order IDs:", df["OrderID"].duplicated().sum())

# Step 5: Clean and transform the data
df = df.drop_duplicates(subset="OrderID", keep="first")
df["Category"] = df["Category"].str.strip().str.title()
df["Region"] = df["Region"].str.strip().str.title()
df["Date"] = pd.to_datetime(df["Date"], errors="coerce")
df["Quantity"] = df["Quantity"].fillna(df["Quantity"].median())
df["UnitPrice"] = df["UnitPrice"].fillna(df["UnitPrice"].median())
df = df.dropna(subset=["Date", "OrderID", "Product", "Category"])
df["TotalSales"] = df["Quantity"] * df["UnitPrice"]
df["Month"] = df["Date"].dt.strftime("%B")

# Save the cleaned unified dataset
df.to_csv("cleaned_sales.csv", index=False)

print("\n--- CLEANED DATA ---")
print(df.to_string(index=False))
print("\nMissing values after cleaning:\n", df.isnull().sum())
print("\nCleaned dataset saved as cleaned_sales.csv")

# Step 6: Perform descriptive and sales analysis
print("\n--- DESCRIPTIVE STATISTICS ---")
print(df[["Quantity", "UnitPrice", "TotalSales"]].describe())

print("\nTotal Sales:", round(df["TotalSales"].sum(), 2))
print("Average Order Value:", round(df["TotalSales"].mean(), 2))
print("Total Orders:", df["OrderID"].nunique())

print("\n--- SALES BY CATEGORY ---")
print(df.groupby("Category")["TotalSales"].sum().sort_values(ascending=False))

print("\n--- SALES BY REGION ---")
print(df.groupby("Region")["TotalSales"].sum())

print("\n--- PRODUCT CATEGORY DISTRIBUTION ---")
print(df["Category"].value_counts())

print("\n--- MONTHLY SALES ---")
print(df.groupby("Month")["TotalSales"].sum())

# Step 7: Visualize the sales data
sns.set_theme(style="whitegrid")

# Bar plot: Total sales by category
plt.figure(figsize=(8, 5))
sns.barplot(data=df, x="Category", y="TotalSales", estimator=sum, errorbar=None)
plt.title("Total Sales by Product Category")
plt.xlabel("Category")
plt.ylabel("Total Sales")
plt.xticks(rotation=15)
plt.tight_layout()
plt.show()

# Pie chart: Distribution of orders by category
category_counts = df["Category"].value_counts()
plt.figure(figsize=(7, 7))
plt.pie(category_counts, labels=category_counts.index,
        autopct="%1.1f%%", startangle=90)
plt.title("Order Distribution by Category")
plt.tight_layout()
plt.show()

# Box plot: Sales distribution by region
plt.figure(figsize=(8, 5))
sns.boxplot(data=df, x="Region", y="TotalSales")
plt.title("Sales Distribution by Region")
plt.xlabel("Region")
plt.ylabel("Total Sales")
plt.tight_layout()
plt.show()

print("\nSales analysis completed successfully!")
