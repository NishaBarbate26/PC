import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from statsmodels.tsa.arima.model import ARIMA

# Step 1: Import the dataset
df = pd.read_csv("Stock_Prices.csv")

print("\n--- FIRST FIVE RECORDS ---")
print(df.head())
print("\nDataset shape:", df.shape)
print("\nColumns:", df.columns.tolist())
print("\nData types:\n", df.dtypes)
print("\nMissing values:\n", df.isnull().sum())

# Step 2: Identify date and closing price columns
df.columns = df.columns.str.strip().str.lower().str.replace(" ", "_")

date_col = next(
    (c for c in ["date", "datetime", "timestamp"] if c in df.columns), None
)
close_col = next(
    (c for c in ["close", "closing_price", "adj_close", "price"] if c in df.columns),
    None
)

if date_col is None or close_col is None:
    raise ValueError("CSV must contain a date column and a closing-price column.")

# Step 3: Convert date and price columns
df[date_col] = pd.to_datetime(df[date_col], errors="coerce")
df[close_col] = pd.to_numeric(df[close_col], errors="coerce")

# Convert other numeric columns where possible
for col in df.columns:
    if col not in [date_col, close_col]:
        df[col] = pd.to_numeric(df[col], errors="ignore")

df = df.dropna(subset=[date_col, close_col])
df = df.drop_duplicates(subset=[date_col])
df = df.sort_values(date_col).set_index(date_col)

if len(df) < 15:
    raise ValueError("At least 15 valid dated stock-price records are recommended.")

print("\n--- CLEANED DATA ---")
print(df.head())
print("\nDate range:", df.index.min(), "to", df.index.max())

# Step 4: Plot historical closing prices
plt.figure(figsize=(10, 5))
plt.plot(df.index, df[close_col], label="Closing Price")
plt.title("Historical Stock Closing Prices")
plt.xlabel("Date")
plt.ylabel("Closing Price")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# Step 5: Calculate and plot moving averages
df["MA_7"] = df[close_col].rolling(window=7).mean()
df["MA_30"] = df[close_col].rolling(window=30).mean()

plt.figure(figsize=(10, 5))
plt.plot(df.index, df[close_col], label="Closing Price", alpha=0.6)
plt.plot(df.index, df["MA_7"], label="7-Record Moving Average")
plt.plot(df.index, df["MA_30"], label="30-Record Moving Average")
plt.title("Stock Prices with Moving Averages")
plt.xlabel("Date")
plt.ylabel("Closing Price")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# Step 6: Analyze weekly and monthly patterns
df["Weekday"] = df.index.day_name()
df["Month"] = df.index.month_name()

weekly = df.groupby("Weekday")[close_col].mean()
weekday_order = [
    "Monday", "Tuesday", "Wednesday", "Thursday", "Friday", "Saturday", "Sunday"
]
weekly = weekly.reindex([d for d in weekday_order if d in weekly.index])

monthly = df.groupby(df.index.month)[close_col].mean()
monthly.index = [
    pd.Timestamp(2000, month, 1).strftime("%b") for month in monthly.index
]

plt.figure(figsize=(9, 5))
weekly.plot(kind="bar", color="teal")
plt.title("Average Closing Price by Weekday")
plt.xlabel("Weekday")
plt.ylabel("Average Closing Price")
plt.xticks(rotation=45)
plt.tight_layout()
plt.show()

plt.figure(figsize=(9, 5))
monthly.plot(kind="bar", color="orange")
plt.title("Average Closing Price by Month")
plt.xlabel("Month")
plt.ylabel("Average Closing Price")
plt.tight_layout()
plt.show()

# Step 7: Analyze correlation with other numeric variables
numeric_df = df.select_dtypes(include=np.number).drop(
    columns=["MA_7", "MA_30"], errors="ignore"
)

if numeric_df.shape[1] >= 2:
    correlation = numeric_df.corr()
    print("\n--- CORRELATION MATRIX ---")
    print(correlation)

    plt.figure(figsize=(8, 6))
    sns.heatmap(correlation, annot=True, cmap="coolwarm", fmt=".2f")
    plt.title("Correlation Between Stock Variables")
    plt.tight_layout()
    plt.show()
else:
    print("\nCorrelation analysis requires at least two numeric columns.")

# Step 8: Forecast closing prices using ARIMA
prices = df[close_col].astype(float)

# Use a simple ARIMA model for forecasting
model = ARIMA(prices, order=(1, 1, 1))
model_fit = model.fit()

forecast_steps = 10
forecast = model_fit.forecast(steps=forecast_steps)

# Create future business-day dates
future_dates = pd.bdate_range(
    start=df.index.max() + pd.Timedelta(days=1),
    periods=forecast_steps
)

forecast_df = pd.DataFrame({
    "Date": future_dates,
    "Forecast_Closing_Price": forecast.to_numpy()
})

print("\n--- NEXT 10 PERIOD FORECAST ---")
print(forecast_df.to_string(index=False))

plt.figure(figsize=(10, 5))
plt.plot(df.index, prices, label="Historical Closing Price")
plt.plot(
    future_dates,
    forecast.to_numpy(),
    "r--o",
    label="ARIMA Forecast"
)
plt.title("Stock Price Forecast Using ARIMA")
plt.xlabel("Date")
plt.ylabel("Closing Price")
plt.legend()
plt.grid(True)
plt.tight_layout()
plt.show()

# Step 9: Save forecast results
forecast_df.to_csv("Stock_Price_Forecast.csv", index=False)
print("\nForecast saved to Stock_Price_Forecast.csv")
print("\nTime series analysis completed successfully!")
