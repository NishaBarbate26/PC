import requests
import csv
from bs4 import BeautifulSoup
from urllib.parse import urljoin

url = "https://books.toscrape.com/"
products_data = []

while url:
    response = requests.get(url, timeout=15)
    response.raise_for_status()

    soup = BeautifulSoup(response.text, "html.parser")
    products = soup.select("article.product_pod")

    for product in products:
        title_link = product.select_one("h3 a")
        price_tag = product.select_one(".price_color")
        stock_tag = product.select_one(".availability")

        title = title_link.get("title", title_link.get_text(strip=True))
        link = urljoin(url, title_link.get("href", ""))
        price = price_tag.get_text(strip=True) if price_tag else "N/A"
        stock = stock_tag.get_text(strip=True) if stock_tag else "N/A"

        products_data.append({
            "Product Name": title,
            "Price": price,
            "Availability": stock,
            "Product Link": link
        })

    next_page = soup.select_one("li.next a")

    if next_page:
        url = urljoin(url, next_page.get("href", ""))
    else:
        url = None

with open("products.csv", "w", newline="", encoding="utf-8") as file:
    fieldnames = ["Product Name", "Price", "Availability", "Product Link"]
    writer = csv.DictWriter(file, fieldnames=fieldnames)
    writer.writeheader()
    writer.writerows(products_data)

print("Web crawling completed successfully!")
print("Total products extracted:", len(products_data))
print("Data saved to products.csv")

for product in products_data[:5]:
    print("\nProduct Name:", product["Product Name"])
    print("Price:", product["Price"])
    print("Availability:", product["Availability"])
    print("Product Link:", product["Product Link"])


# LINE-BY-LINE EXPLANATION:
# import requests: Downloads web pages.
# import csv: Creates a CSV file for storing product details.
# BeautifulSoup: Parses HTML content.
# urljoin: Converts relative links into complete URLs.
# url: Stores the starting e-commerce website address.
# products_data: Stores the extracted product information.
# while url: Continues crawling while another page is available.
# requests.get(): Sends a request to download the webpage.
# timeout=15: Stops waiting if the request takes too long.
# raise_for_status(): Reports HTTP errors.
# BeautifulSoup(): Converts downloaded HTML into a searchable structure.
# soup.select(): Finds all product containers on the current page.
# for product: Processes each product separately.
# select_one("h3 a"): Finds the product title link.
# select_one(".price_color"): Finds the product price.
# select_one(".availability"): Finds the stock information.
# get("title", ...): Gets the full product title or uses visible text.
# get_text(strip=True): Extracts text without extra whitespace.
# urljoin(): Builds the complete product URL.
# if price_tag else: Handles a missing price safely.
# if stock_tag else: Handles missing availability information.
# products_data.append(): Stores the product details in a dictionary.
# soup.select_one("li.next a"): Finds the next-page link.
# if next_page: Checks whether another page exists.
# urljoin(): Builds the URL of the next catalogue page.
# url = None: Stops the crawler when no next page exists.
# open(): Creates products.csv for writing.
# newline="": Prevents unwanted blank lines in the CSV file.
# encoding="utf-8": Supports a wide range of text characters.
# fieldnames: Defines the CSV column headings.
# csv.DictWriter(): Writes dictionary records into CSV format.
# writeheader(): Writes the column names.
# writerows(): Writes all extracted product records.
# print(): Displays the crawling status and record count.
# products_data[:5]: Selects the first five products for display.
# The final print statements display each product's details.
