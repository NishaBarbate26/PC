
# Practical 3: Quantum Production System for the 3-Puzzle Problem
# Educational implementation using puzzle production rules,
# classical BFS verification, and a Grover search demonstration.
#
# VS CODE SETUP AND EXECUTION COMMANDS:
# 1. Install Python 3.10 or later.
# 2. Open the terminal in VS Code.
# 3. Install Qiskit:

# 5. Run:
#    python quantum_3_puzzle.py

from collections import deque
from qiskit import QuantumCircuit
from qiskit.quantum_info import Statevector


# Step 1: Define the initial and goal states.
# Zero represents the blank tile.

initial = (
    1, 2, 0,
    3, 4, 5,
    6, 7, 8
)

goal = (
    1, 2, 3,
    4, 5, 6,
    7, 8, 0
)

# Step 2: Define production rules.
# Each rule specifies a legal movement of the blank tile.

moves = {
    "U": -3,
    "D": 3,
    "L": -1,
    "R": 1
}


# Step 3: Apply a production rule to a puzzle state.

def apply_move(state, move):
    blank = state.index(0)
    row, col = divmod(blank, 3)

    if move == "U" and row == 0:
        return None
    if move == "D" and row == 2:
        return None
    if move == "L" and col == 0:
        return None
    if move == "R" and col == 2:
        return None

    new_pos = blank + moves[move]
    new_state = list(state)

    new_state[blank], new_state[new_pos] = (
        new_state[new_pos],
        new_state[blank]
    )

    return tuple(new_state)


# Step 4: Display the puzzle board.

def display_board(state):
    for i in range(0, 9, 3):
        print(" ".join(
            "_" if tile == 0 else str(tile)
            for tile in state[i:i + 3]
        ))
    print()


# Step 5: Implement the production-system search.
# BFS finds a shortest legal sequence of production rules.

def solve_puzzle(start, target):
    queue = deque([(start, [])])
    visited = {start}

    while queue:
        state, path = queue.popleft()

        # Goal test.
        if state == target:
            return path

        # Apply each production rule.
        for move in moves:
            next_state = apply_move(state, move)

            if next_state is not None and next_state not in visited:
                visited.add(next_state)
                queue.append((next_state, path + [move]))

    return None


# Step 6: Demonstrate Grover quantum search.
# This separate two-qubit demonstration searches four move labels.
# Encoding: 00=U, 01=D, 10=L, 11=R.
# The oracle marks label 11; it does NOT evaluate the puzzle goal.

def grover_search_demo():
    qc = QuantumCircuit(2)

    # Prepare equal superposition.
    qc.h([0, 1])

    # Oracle: mark the basis state |11>.
    qc.cz(0, 1)

    # Grover diffusion operator.
    qc.h([0, 1])
    qc.x([0, 1])
    qc.h(1)
    qc.cz(0, 1)
    qc.h(1)
    qc.x([0, 1])
    qc.h([0, 1])

    return qc


# Step 7: Execute the puzzle production system.

if __name__ == "__main__":
    print("3-Puzzle Production System")
    print("=" * 35)

    print("\nInitial State:")
    display_board(initial)

    print("Goal State:")
    display_board(goal)

    solution = solve_puzzle(initial, goal)

    if solution is None:
        print("No solution exists for the given states.")
    else:
        print("Shortest solution:", " -> ".join(solution))
        print("Number of moves:", len(solution))

        current_state = initial

        for step, move in enumerate(solution, start=1):
            current_state = apply_move(current_state, move)

            print(f"\nStep {step}: Apply rule {move}")
            display_board(current_state)

        print("Goal reached:", current_state == goal)

    # Step 8: Run the Grover demonstration.
    print("\nGrover Quantum Search Demonstration")
    print("=" * 35)

    qc = grover_search_demo()

    print("\nQuantum Circuit:")
    print(qc.draw(output="text"))

    statevector = Statevector.from_instruction(qc)

    print("\nMeasurement probabilities:")
    print(statevector.probabilities_dict())

    print("\nNote:")
    print("The puzzle is solved using classical BFS.")
    print("The Grover circuit is a separate demonstration.")
    print("This is not a complete implementation of")
    print("Tarrataca's original quantum production-system algorithm.")
