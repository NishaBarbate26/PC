import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

df = pd.read_csv("Iris.csv")
df.columns = df.columns.str.strip()

print("First Five Rows:")
print(df.head())

X = df.select_dtypes(include="number")
X = X.drop(columns=["Id"], errors="ignore")

X_scaled = StandardScaler().fit_transform(X)

inertia = []

for k in range(1, 11):
    model = KMeans(n_clusters=k, random_state=42, n_init=10)
    model.fit(X_scaled)
    inertia.append(model.inertia_)

plt.plot(range(1, 11), inertia, marker="o")
plt.xlabel("Number of Clusters (K)")
plt.ylabel("WCSS / Inertia")
plt.title("Elbow Method for K-Means")
plt.xticks(range(1, 11))
plt.grid(True)
plt.tight_layout()
plt.show()

optimal_k = 3

kmeans = KMeans(n_clusters=optimal_k, random_state=42, n_init=10)
df["Cluster"] = kmeans.fit_predict(X_scaled)

print("\nOptimal Number of Clusters:", optimal_k)
print("\nCluster Counts:")
print(df["Cluster"].value_counts().sort_index())

if X_scaled.shape[1] >= 2:
    plt.scatter(
        X_scaled[:, 0], X_scaled[:, 1],
        c=df["Cluster"], cmap="viridis"
    )
    plt.xlabel(X.columns[0])
    plt.ylabel(X.columns[1])
    plt.title("K-Means Clustering of Iris Dataset")
    plt.tight_layout()
    plt.show()

print("\nClustered Data:")
print(df.head())


# LINE-BY-LINE EXPLANATION:
# pandas: Loads and manages the Iris dataset.
# matplotlib: Displays the elbow curve and cluster plot.
# StandardScaler: Standardizes numerical features.
# KMeans: Performs K-Means clustering.
# read_csv(): Loads Iris.csv.
# select_dtypes(): Selects numerical columns.
# drop(): Removes the optional Id column.
# fit_transform(): Standardizes the dataset.
# inertia: Stores WCSS values for different cluster counts.
# n_clusters=k: Creates k clusters.
# inertia_: Measures within-cluster sum of squared distances.
# The elbow graph helps identify a suitable cluster count.
# optimal_k=3: Uses three clusters for the Iris dataset.
# fit_predict(): Assigns each sample to a cluster.
# value_counts(): Counts samples in each cluster.
# scatter(): Visualizes clusters using two features.
