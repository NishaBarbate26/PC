import nltk
import re
from nltk.corpus import stopwords
from nltk.stem import PorterStemmer

nltk.download('stopwords', quiet=True)

text = """The students are learning and studying different subjects.
They were playing games and running quickly."""

text = text.lower()

text = re.sub(r'[^a-z\s]', '', text)

words = text.split()

stop_words = set(stopwords.words('english'))

filtered_words = [word for word in words if word not in stop_words]

stemmer = PorterStemmer()

stemmed_words = [stemmer.stem(word) for word in filtered_words]

print("Original Text:\n", text)
print("\nWords after Stop Word Removal:\n", filtered_words)
print("\nWords after Stemming:\n", stemmed_words)

# from nltk.stem import PorterStemmer: Imports the stemming algorithm.

# text = ...: Stores the sample text document.
# text.lower(): Converts all letters into lowercase.
# re.sub(): Removes punctuation, digits, and special characters.
# text.split(): Splits the text into individual words.
# stopwords.words(): Loads common English stop words.
# set(): Stores stop words for efficient lookup.
