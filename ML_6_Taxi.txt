import gymnasium as gym
import numpy as np

# Create the Taxi environment
env = gym.make("Taxi-v3")

# Initialize the Q-table
q_table = np.zeros((env.observation_space.n, env.action_space.n))

# Set learning parameters
learning_rate = 0.7
discount_factor = 0.9
epsilon = 1.0
epsilon_decay = 0.995
min_epsilon = 0.01
episodes = 5000

# Train the agent
for episode in range(episodes):
    state, info = env.reset()
    terminated = False
    truncated = False

    while not (terminated or truncated):
        # Choose an action using epsilon-greedy policy
        if np.random.random() < epsilon:
            action = env.action_space.sample()
        else:
            action = np.argmax(q_table[state])

        next_state, reward, terminated, truncated, info = env.step(action)

        # Update the Q-table using Q-learning
        best_next_action = np.max(q_table[next_state])
        q_table[state, action] += learning_rate * (
            reward
            + discount_factor * best_next_action
            - q_table[state, action]
        )

        state = next_state

    epsilon = max(min_epsilon, epsilon * epsilon_decay)

# Test the trained agent
total_rewards = []

for episode in range(5):
    state, info = env.reset()
    terminated = False
    truncated = False
    total_reward = 0
    steps = 0

    while not (terminated or truncated):
        action = np.argmax(q_table[state])
        state, reward, terminated, truncated, info = env.step(action)

        total_reward += reward
        steps += 1

    total_rewards.append(total_reward)
    print(
        f"Episode {episode + 1}: "
        f"Steps = {steps}, Reward = {total_reward}"
    )

print("\nAverage reward:", np.mean(total_rewards))

env.close()

# Explanation:
# 1. Create the Taxi-v3 environment with passenger and destination locations.
# 2. Initialize a Q-table to store the value of each action in every state.
# 3. Train the agent using exploration and exploitation.
# 4. Update Q-values using the Q-learning formula after each action.
# 5. Reduce epsilon over time so the agent explores less and uses learned actions.
# 6. Test the trained agent and display steps and rewards for five episodes.
